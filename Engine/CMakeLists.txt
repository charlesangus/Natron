# Native node registration fragments (see Engine/Nodes/NativeRegistry.h), aggregated into
# NativeRegistryAll.inc. Sorted so the registration order does not depend on the filesystem.
# Each fragment is a configure dependency because the header it names is copied into the
# aggregate.
file(GLOB_RECURSE NatronEngineNodes_REGISTRATIONS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Nodes/*.reg.inc)
list(SORT NatronEngineNodes_REGISTRATIONS)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${NatronEngineNodes_REGISTRATIONS})
get_filename_component(NATRON_SOURCE_ROOT .. ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(nativeRegistryHeaders "")
set(nativeRegistryNodes "")
foreach(fragment IN LISTS NatronEngineNodes_REGISTRATIONS)
    file(READ ${fragment} fragmentText)
    string(REGEX MATCHALL "NATRON_NATIVE_NODE\\(" fragmentNodes "${fragmentText}")
    list(LENGTH fragmentNodes fragmentNodeCount)
    if(NOT fragmentNodeCount EQUAL 1
       OR NOT fragmentText MATCHES "NATRON_NATIVE_NODE\\([^\n]+,[ \t]*\"([^\"]+)\"[ \t]*\\)")
        message(FATAL_ERROR "${fragment}: a native registration fragment holds exactly one NATRON_NATIVE_NODE(Class, \"Header.h\") line")
    endif()
    get_filename_component(fragmentDir ${fragment} DIRECTORY)
    set(fragmentHeader ${fragmentDir}/${CMAKE_MATCH_1})
    if(NOT EXISTS ${fragmentHeader})
        message(FATAL_ERROR "${fragment}: names ${CMAKE_MATCH_1}, which does not exist relative to it")
    endif()
    file(RELATIVE_PATH fragmentHeader ${NATRON_SOURCE_ROOT} ${fragmentHeader})
    file(RELATIVE_PATH fragmentPath ${NATRON_SOURCE_ROOT} ${fragment})
    list(APPEND nativeRegistryHeaders "#include \"${fragmentHeader}\"\n")
    list(APPEND nativeRegistryNodes "#include \"${fragmentPath}\"\n")
endforeach()
list(REMOVE_DUPLICATES nativeRegistryHeaders)
string(JOIN "" nativeRegistryHeaders ${nativeRegistryHeaders})
string(JOIN "" nativeRegistryNodes ${nativeRegistryNodes})
# Written through configure_file() so an unchanged aggregate keeps its timestamp and does not
# rebuild AppManager.cpp.
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/NativeRegistryAll.inc.tmp
    "// Generated by Engine/CMakeLists.txt from Engine/Nodes/**/*.reg.inc. Do not edit.\n"
    "#ifdef NATRON_NATIVE_REGISTRY_HEADERS\n"
    "${nativeRegistryHeaders}"
    "#else\n"
    "${nativeRegistryNodes}"
    "#endif\n")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/NativeRegistryAll.inc.tmp ${CMAKE_CURRENT_BINARY_DIR}/NativeRegistryAll.inc COPYONLY)

# GPU cost model seed fragments (see Engine/GpuCostModel.cpp), one per native node beside its
# registration fragment and one per OFX plug-in, aggregated into GpuCostSeedsAll.inc. The
# aggregate defines each row macro its includer leaves undefined as empty, so every table
# collects only its own rows, and an empty glob is valid.
file(GLOB_RECURSE NatronEngine_COST_SEEDS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Nodes/*.costseed.inc)
file(GLOB NatronEngine_OFX_COST_SEEDS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/GpuCostSeeds/ofx/*.inc)
list(APPEND NatronEngine_COST_SEEDS ${NatronEngine_OFX_COST_SEEDS})
list(SORT NatronEngine_COST_SEEDS)
set(costSeedMacros
    NATRON_GPU_COST_SEED_CPU_ROW
    NATRON_GPU_COST_SEED_CPU_MOTION_BLUR_ROW
    NATRON_GPU_COST_SEED_CPU_SIGMA_ROW
    NATRON_GPU_COST_SEED_GPU_ROW)
set(costSeedDefaults "")
set(costSeedUndefs "")
foreach(macro IN LISTS costSeedMacros)
    string(APPEND costSeedDefaults "#ifndef ${macro}\n#define ${macro}(...)\n#endif\n")
    string(APPEND costSeedUndefs "#undef ${macro}\n")
endforeach()
set(costSeedIncludes "")
foreach(fragment IN LISTS NatronEngine_COST_SEEDS)
    file(RELATIVE_PATH fragmentPath ${NATRON_SOURCE_ROOT} ${fragment})
    string(APPEND costSeedIncludes "#include \"${fragmentPath}\"\n")
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/GpuCostSeedsAll.inc.tmp
    "// Generated by Engine/CMakeLists.txt from Engine/Nodes/**/*.costseed.inc and\n"
    "// Engine/GpuCostSeeds/ofx/*.inc. Do not edit.\n"
    "${costSeedDefaults}"
    "${costSeedIncludes}"
    "${costSeedUndefs}")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/GpuCostSeedsAll.inc.tmp ${CMAKE_CURRENT_BINARY_DIR}/GpuCostSeedsAll.inc COPYONLY)

# Retired OFX plug-ins (see Engine/OfxRetired/README.md and Engine/OfxRetired.h): one marker file
# per retired ID, aggregated into OfxRetiredAll.inc as a constexpr array sorted by ID. Each marker
# is a configure dependency because its content, the replacing native ID, is copied into the
# aggregate. An empty directory yields a zero-sized std::array, never a zero-length C array.
file(GLOB NatronEngine_OFX_RETIRED LIST_DIRECTORIES false CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/OfxRetired/*)
list(FILTER NatronEngine_OFX_RETIRED EXCLUDE REGEX "/README\\.md$")
list(SORT NatronEngine_OFX_RETIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${NatronEngine_OFX_RETIRED})
set(ofxPluginIdRegex "^[A-Za-z0-9_-]+(\\.[A-Za-z0-9_-]+)+$")
set(nativePluginIdRegex "^[A-Za-z0-9_-]+(\\.[A-Za-z0-9_-]+)*$")
set(ofxRetiredRows "")
list(LENGTH NatronEngine_OFX_RETIRED ofxRetiredCount)
foreach(marker IN LISTS NatronEngine_OFX_RETIRED)
    get_filename_component(retiredId ${marker} NAME)
    if(NOT retiredId MATCHES "${ofxPluginIdRegex}")
        message(FATAL_ERROR "${marker}: a retired OFX marker is named after the plug-in ID it retires, e.g. net.sf.openfx.SharpenPlugin")
    endif()
    file(READ ${marker} replacement)
    string(STRIP "${replacement}" replacement)
    if(NOT replacement STREQUAL "" AND NOT replacement MATCHES "${nativePluginIdRegex}")
        message(FATAL_ERROR "${marker}: a retired OFX marker holds at most one line, the ID of the native plug-in that replaces it")
    endif()
    string(APPEND ofxRetiredRows "    { \"${retiredId}\", \"${replacement}\" },\n")
endforeach()
if(ofxRetiredCount EQUAL 0)
    set(ofxRetiredInit "{}")
else()
    set(ofxRetiredInit "{ {\n${ofxRetiredRows}} }")
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/OfxRetiredAll.inc.tmp
    "// Generated by Engine/CMakeLists.txt from Engine/OfxRetired/*. Do not edit.\n"
    "constexpr std::array<RetiredOfxMarker, ${ofxRetiredCount}> kRetiredOfxMarkers = ${ofxRetiredInit};\n")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/OfxRetiredAll.inc.tmp ${CMAKE_CURRENT_BINARY_DIR}/OfxRetiredAll.inc COPYONLY)

# Some Linux dependencies (notably Qt) can expose the distribution FreeType
# before this static library's transitive link interface is expanded.  The
# ASWF image also provides HarfBuzz and a newer, ABI-compatible FreeType under
# /usr/local; loading the distribution library first wins by SONAME and leaves
# HarfBuzz's newer FT_* references unresolved.  Promote this ordered pair to
# direct dependencies of every final NatronEngine consumer so the matching
# /usr/local FreeType is selected before any indirect system copy.
if(UNIX AND NOT APPLE AND CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
    set_property(TARGET NatronEngine APPEND PROPERTY
        INTERFACE_LINK_LIBRARIES_DIRECT
        harfbuzz::harfbuzz
        Freetype::Freetype
    )
endif()
